cmake_minimum_required(VERSION 3.22.1)
project(sandev_llama LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
    message(FATAL_ERROR "llama.cpp source is missing. The GitHub Actions workflow must fetch the pinned llama.cpp source before building.")
endif()

add_subdirectory(llama.cpp)
add_library(sandev_llama SHARED llama_jni.cpp)
target_compile_features(sandev_llama PRIVATE cxx_std_17)
target_link_libraries(sandev_llama PRIVATE llama android log)
